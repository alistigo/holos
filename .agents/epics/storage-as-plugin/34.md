```typescript
import type { AlistigoPlugin } from "@alistigo/artifact-plugin-api";
import { ClaudeArtifactListRepository } from "./claude-artifact-list-repository.js";
import { isClaudeArtifactContext } from "./claude-artifact-list-repository.js";

// Re-use the withStorageTimeout helper from claude-artifact-list-repository.ts
// (move to a shared internal file if needed, or inline it here)

const claudeStoragePlugin: AlistigoPlugin = {
  name: "@alistigo/claude-storage-plugin",
  type: "storage",
  storage: {
    isAvailable(): boolean {
      return isClaudeArtifactContext();
    },
    createStore() {
      return new ClaudeArtifactListRepository();
    },
    async listKeys(prefix = ""): Promise<Array<{ key: string; value: string }>> {
      const storage = (window as Window & { storage?: ClaudeStorage }).storage;
      if (!storage) return [];
      try {
        const results = await withStorageTimeout(storage.list(prefix));
        return (results as Array<{ key: string; value: string }>).map(({ key, value }) => ({ key, value }));
      } catch {
        return [];
      }
    },
    // No seedIfEmpty — Claude storage persists across sessions natively
  },
};

export default claudeStoragePlugin;
```

Note: `withStorageTimeout` is defined in `claude-artifact-list-repository.ts` — either keep it there and import it, or extract to `src/utils.ts`.

**New file `src/hooks/useLocalStorageEntries.ts`:**
```typescript
import localStoragePlugin from "@alistigo/local-storage-plugin";
import { useCallback, useEffect, useState } from "react";

const PREFIX = "alistigo";

async function fetchEntries(): Promise<Array<{ key: string; value: string }>> {
  if (!localStoragePlugin.storage?.isAvailable()) return [];
  return localStoragePlugin.storage.listKeys(PREFIX);
}

export function useLocalStorageEntries() {
  const [entries, setEntries] = useState<Array<{ key: string; value: string }>>([]);

  const refresh = useCallback(() => {
    void fetchEntries().then(setEntries);
  }, []);

  useEffect(() => {
    refresh();
    // storage event fires when another frame of same origin writes to localStorage
    window.addEventListener("storage", refresh);
    return () => window.removeEventListener("storage", refresh);
  }, [refresh]);

  return { entries, refresh };
}
```
